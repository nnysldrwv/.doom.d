** Portable tool locations (Windows)
Git for Windows 可能装在 =C:\Program Files\Git=，也可能由 scoop 装在
=~\scoop\apps\git\current=。配置里多处需要它的 GNU 用户态工具
(=find.exe= / =unzip.exe= / =bash.exe=)，这里统一探测，避免硬编码路径或用户名。
#+begin_src emacs-lisp
(defvar my/git-usr-bin
  (catch 'found
    (dolist (candidate (list (expand-file-name "scoop/apps/git/current/usr/bin" "~")
                             "C:/Program Files/Git/usr/bin"
                             "C:/PROGRA~1/Git/usr/bin"))
      (when (file-directory-p candidate)
        (throw 'found candidate))))
  "Directory holding Git for Windows' GNU userland (find/unzip/bash).")

(defun my/git-tool (name)
  "Return the full path to Git for Windows' GNU tool NAME, or nil."
  (when my/git-usr-bin
    (let ((file (expand-file-name name my/git-usr-bin)))
      (when (file-executable-p file) file))))

** Windows: 子进程参数的编码（ANSI 代码页）
在 Windows 上，Emacs 用「进程的输入编码系统」——也就是
=default-process-coding-system= 的 =car=——来编码 =call-process= 的参数，然后把
命令行交给 =CreateProcessA=；Windows 侧再按 ANSI 代码页（=w32-ansi-code-page=，
简中系统是 936/GBK）解码回 Unicode。

本文件上方执行了 =(prefer-coding-system 'utf-8)=，它把
=default-process-coding-system= 变成了 =(utf-8-dos . utf-8-unix)=。于是含中文的
路径被编成 UTF-8 字节、又被 Windows 当 GBK 解码，子进程拿到的是乱码路径。
典型症状：=unzip= 报 =cannot find or open <乱码>.epub= 并以退出码 9 失败，而
同样的文件换成 ASCII 文件名就正常。

修法：调用这类外部程序时，把「写向进程的编码」临时切回 ANSI 代码页对应的编码，
用 =my/with-ansi-process-coding= 包一层即可；非 Windows 上是 no-op。

注意：Emacs 编码 =call-process= 参数时用的是「写向进程」的那套编码——也就是
=default-process-coding-system= 的 **=cdr=**（不是 =car=！实测：=(gbk . utf-8)=
传参仍是 UTF-8 会失败，=(utf-8 . gbk)= 则成功），或者被绑定的
=coding-system-for-write=。所以这里绑定 =coding-system-for-write=，只动写侧，
读侧（进程输出怎么解码）继续沿用全局的 utf-8 设置。
#+begin_src emacs-lisp
(defun my/w32-ansi-coding-system ()
  "Return the coding system matching `w32-ansi-code-page'."
  (let ((cs (pcase (and (boundp 'w32-ansi-code-page) w32-ansi-code-page)
              ((or 936 54936) 'gbk)
              (950 'big5)
              (932 'cp932)
              (949 'cp949)
              (_ 'utf-8))))          ; 含 65001（系统开了 UTF-8 全局 beta）
    (if (coding-system-p cs) cs 'utf-8)))

(defmacro my/with-ansi-process-coding (&rest body)
  "Run BODY with `coding-system-for-write' bound to the Windows ANSI code page.
Wrap `call-process' calls that pass non-ASCII (e.g. Chinese) file names as
arguments: Emacs encodes those arguments with the coding system used to write
to the process, and Windows then decodes the command line with CP_ACP.

Binding only the write side keeps the read side (how process output is
decoded) at the global setting.  No-op on non-Windows systems."
  (declare (indent 0))
  (if (featurep :system 'windows)
      `(let ((coding-system-for-write (my/w32-ansi-coding-system)))
         ,@body)
    `(progn ,@body)))
#+end_src

(defun my/org-html-to-org (html img-dir)
  "Convert HTML string to Org with pandoc, extracting images into IMG-DIR.
Returns the Org text, or nil on failure."
  (unless (executable-find "pandoc")
    (user-error "pandoc not found (sudo apt install pandoc)"))
  (make-directory img-dir t)
  (let* ((tmp (make-temp-file "org-clip" nil ".html"))
         (out (concat (file-name-sans-extension tmp) ".org")))
    (unwind-protect
        (progn
          (let ((coding-system-for-write 'utf-8))
            (write-region html nil tmp nil 'silent))
          (and (zerop (call-process "pandoc" nil nil nil
                                    ;; auto_identifiers 会给每个标题加 :PROPERTIES: 抽屉, 关掉更干净
                                    "-f" "html-auto_identifiers" "-t" "org"
                                    "--wrap=none" (concat "--extract-media=" img-dir)
                                    tmp "-o" out))
               (with-temp-buffer
                 (let ((coding-system-for-read 'utf-8))
                   (insert-file-contents out))
                 ;; pandoc 抽出媒体时写的是绝对路径, 改成相对路径免得换机器就断链
                 (let ((text (string-trim (buffer-string))))
                   (replace-regexp-in-string
                    (regexp-quote (file-name-unquote img-dir))
                    (file-relative-name img-dir default-directory)
                    text t t)))))
      (ignore-errors (delete-file tmp))
      (ignore-errors (delete-file out)))))

(defun my/org-image-caption ()
  "Add a #+CAPTION line above the image at point, leaving point at its end.
Re-run on an already-captioned image to append to the existing caption.
Implemented with plain buffer scanning instead of `org-element-context',
whose  :begin  points to the wrong line right after the buffer was edited."
  (interactive)
  (unless (my/org-image-file-at-point)
    (user-error "No image file at point"))
  (let ((beg (line-beginning-position))
        ;; 坑: '#+' 里的 + 是字面字符, 必须转义成 #\\+, 否则被当成量词永远匹配不上
        (re  "#\\+\\(CAPTION\\|ATTR_\\|NAME\\|HEADER\\)"))
    ;; 向上吞掉连续的 affiliated 关键字行, 得到整个块的起始行。
    ;; 坑: forward-line 会试图保留列位置, 每次之后都要显式 beginning-of-line。
    (save-excursion
      (goto-char beg)
      (forward-line -1)
      (beginning-of-line)
      (while (and (not (bobp)) (looking-at-p re))
        (setq beg (point))

(defun my/dnd--local-path (uri)
  "Return the local file path for a dropped URI, or nil.

`dnd-get-local-file-name' turns the XDND form \"file:///C:/foo\" into
\"/C:/foo\", which is NOT a valid path on MS-Windows.  Emacs 30+ rewrote
the w32 drag-and-drop backend and now emits that form, so drops silently
did nothing.  Strip the spurious leading slash(es) so both \"/C:/foo\" and
\"//C:/foo\" become \"C:/foo\".  UNC paths (//server/share/...) are left
alone because they have no drive letter."
  (let ((f (ignore-errors (dnd-get-local-file-name uri nil))))
    (when (and f (eq system-type 'windows-nt))
      (when (string-match "\\`//?\\([A-Za-z]:\\)" f)
        (setq f (replace-match "\\1" nil nil f))))
    f))

(defun my/dnd-fallback (uri action skip)
  "Hand URI back to the normal dnd dispatch, ignoring handler SKIP.

A dnd handler consumes its URI the moment it is called, even when it
returns nil -- so a handler that declines to act MUST re-dispatch, or the
drop silently does nothing.  SKIP is our own handler, excluded to avoid
re-entering ourselves."
  (let ((alist nil))
    (dolist (cell dnd-protocol-alist)
      (unless (eq (cdr cell) skip)
        (push cell alist)))
    (let ((dnd-protocol-alist (nreverse alist)))
      (dnd-handle-multiple-urls (selected-window) (list uri) action))))

(defun my/org-dnd-copy-to-data (uri action)
  "Copy locally-dropped file URI into ~/org/data/ with a denote-style name.
Prompts for denote keywords interactively.  Outside org-mode, or when the
URI is not a real local file, hand the drop back to the normal dispatch."
  (let ((src (and (derived-mode-p 'org-mode) (my/dnd--local-path uri))))
    (if (and src (file-regular-p src))
        (let* ((keywords (denote-keywords-prompt))
               (dest-dir  (expand-file-name "data/" org-directory))
               (dest-name (my/org-download-denote-file-format
                           (file-name-nondirectory src) keywords))
               (width     (or (bound-and-true-p org-download-image-org-width) 800))
               (dest      (expand-file-name dest-name dest-dir)))
          (make-directory dest-dir t)
          (copy-file src dest 1)
          (unless (zerop width)
            (insert (format "#+attr_org: :width %dpx\n" width)))
          (insert (format "[[file:%s]]"
                          (file-relative-name dest default-directory)))
          'copy)
      (my/dnd-fallback uri action #'my/org-dnd-copy-to-data))))

** 明文兜底：非 PGP 数据的日记文件也能打开
~epa-file-name-regexp~ 把 journal/ 下所有 ~__journal.org~ 一律当成加密文件，
只要文件名匹配就无条件走 ~epg-decrypt-file~。于是任何在 Emacs 之外写出来的
明文文件（Trilium 同步、git pull、备份还原、新建未保存）一打开就报
~Can't decrypt, No data (no armored data)~，连内容都看不到，更没法修。
这里在读取前先看一眼文件头：确认不是 OpenPGP 数据就退回普通插入。
明文打开后正常保存，写盘时仍会按名字加密，文件自动回到加密态。
#+begin_src emacs-lisp
(defun my/file-pgp-data-p (file)
  "Return non-nil if FILE starts with OpenPGP data (armored or binary)."
  (and (file-readable-p file)
       (> (file-attribute-size (file-attributes file)) 0)
       (with-temp-buffer
         (set-buffer-multibyte nil)
         (insert-file-contents-literally file nil 0 300)
         (let ((head (buffer-string)))
           (or (string-match-p "-----BEGIN PGP" head)
               ;; 二进制 OpenPGP 旧格式 CTB：bit7 置位、bit6 清零。
               ;; UTF-8 文本首字节不会落在 0x80-0xBF，判断是安全的。
               (= (logand (aref head 0) #xC0) #x80))))))

(defun my/epa-file-insert-file-contents-advice (orig file &optional visit beg end replace)
  "Insert FILE as plain text when it holds no OpenPGP data.
Otherwise defer to `epa-file-insert-file-contents'."
  (if (and (file-exists-p file)
           (not (my/file-pgp-data-p file)))
      (epa-file-run-real-handler #'insert-file-contents
                                 (list file visit beg end replace))
    (funcall orig file visit beg end replace)))

(with-eval-after-load 'epa-file
  ;; `epa-file-run-real-handler' 是宏，必须在本文件加载后才展开。
  (advice-add 'epa-file-insert-file-contents :around
              #'my/epa-file-insert-file-contents-advice))

#+begin_src emacs-lisp
(defvar my/weather-location "上海"
  "wttr.in 查询地点。")

(defvar my/weather--cache nil
  "(日期 . 天气字符串) 缓存, 同一天不再重复请求 wttr.in。")

(defun my/weather--url ()
  "返回 wttr.in 单行天气的 URL。"
  (format "https://wttr.in/%s?format=3"
          (url-hexify-string my/weather-location)))

(defun my/get-weather ()
  "同步获取天气 (手动调用 `my/insert-weather' 时用)。"
  (let ((coding-system-for-read 'utf-8))
    (string-trim
     (shell-command-to-string
      (format "curl -s --connect-timeout 5 --max-time 10 \"%s\""
              (my/weather--url))))))

(defun my/insert-weather ()
  "在光标处插入当前天气 (同步, 可能卡 ≤10s)。"
  (interactive)
  (insert (my/get-weather)))

(defun my/weather--fetch (callback &optional attempt)
  "异步取天气, 非空结果交给 CALLBACK。空结果或 curl 报错时重试一次。

超时放宽到 10s: wttr.in 在国内经常 3s 拿不到, 旧版 --max-time 3
是大部分日记没有天气的直接原因。"
  (let* ((attempt (or attempt 1))
         (buf (generate-new-buffer " *weather*")))
    (make-process
     :name "weather-fetch"
     :buffer buf
     :command (list "curl" "-sS" "--connect-timeout" "5" "--max-time" "10"
                    (my/weather--url))
     :sentinel
     (lambda (proc _event)
       (when (memq (process-status proc) '(exit signal))
         (let ((output (with-current-buffer buf
                         (string-trim (buffer-string)))))
           (kill-buffer buf)
           (cond
            ;; curl 失败时 -sS 会把错误写进 stdout, 别当成天气写进日记。
            ((or (string-empty-p output)
                 (string-prefix-p "curl" output))
             (when (< attempt 2)
               (my/weather--fetch callback (1+ attempt))))
            (t
             (setq my/weather--cache
                   (cons (format-time-string "%Y-%m-%d") output))
             (funcall callback output)))))))))

(defun my/weather--get (callback)
  "取天气: 当天缓存命中就直接回调, 否则发起异步请求。"
  (let ((today (format-time-string "%Y-%m-%d")))
    (if (and my/weather--cache
             (string= (car my/weather--cache) today))
        (funcall callback (cdr my/weather--cache))
      (my/weather--fetch callback))))

(defun my/journal--insert-weather (buffer text)
  "在 BUFFER 的 front matter 之后插入 `* 天气' 标题 + TEXT。

已有天气标题时不重复插入, 返回是否插入。"
  (with-current-buffer buffer
    (let ((m (point-marker))
          inserted)
      (set-marker-insertion-type m t)
      (save-excursion
        (goto-char (point-min))
        (unless (re-search-forward "^\\*+ 天气[ \t]*$" nil t)
          (goto-char (point-min))
          (while (looking-at-p "^#\\+")
            (forward-line 1))
          (while (looking-at-p "^[ \t]*$")
            (forward-line 1))
          (insert "* 天气\n" text "\n\n")
          (setq inserted t)))
      (goto-char m)
      inserted)))

(defun my/denote-journal-auto-weather ()
  "新建 journal 时异步在顶部插入天气, 不阻塞新建流程。

关键点: 只改新建 note 的 buffer。此时内容在内存里仍是明文, 保存时才
按 `epa-file-name-regexp' 加密。旧实现是 curl 回来后 `find-file-noselect'
读文件——那时文件多半已经落盘成密文, 读它等于在异步哨兵里解密,
要么静默失败, 要么等口令把 Emacs 卡住。journal 全量加密一上 (9 月),
天气就基本不再出现了, 时间线对得上。"
  (when (and (buffer-file-name)
             (string-match-p "__journal" (buffer-file-name)))
    (let ((buffer (current-buffer))
          (file (buffer-file-name)))
      (my/weather--get
       (lambda (text)
         (if (buffer-live-p buffer)
             (with-current-buffer buffer
               (when (my/journal--insert-weather buffer text)
                 (save-buffer)))
           ;; buffer 已关: 磁盘上仍是明文才补写; 已加密的文件不碰,
           ;; 免得又要口令。
           (when (and (file-exists-p file)
                      (not (my/file-pgp-data-p file)))
             (with-current-buffer (find-file-noselect file)
               (when (my/journal--insert-weather (current-buffer) text)
                 (save-buffer))))))))))

        (buffer (current-buffer)))

               (not (my/denote-journal--has-weather-p buffer)))

           (with-current-buffer buffer
             (when (and (not (my/denote-journal--has-weather-p buffer))
                        (my/journal--insert-weather buffer text))
               (save-buffer)

(defun my/beancount--parse-balance-batch-line (line line-number broker-account)
  "Parse LINE at LINE-NUMBER into a balance posting plist.
ETF lines use BROKER-ACCOUNT unless LINE starts with a broker account alias."
  (let ((trimmed (string-trim line)))
    (unless (or (string-empty-p trimmed)
                (string-prefix-p "#" trimmed)
                (string-prefix-p ";" trimmed))
      (let (line-broker-account code shares)
        (cond
         ((and (string-match "\\`\\([^[:space:],，]+\\)[[:space:],，]+\\([^[:space:],，]+\\)[[:space:],，]+\\(.+\\)\\'" trimmed)
               (setq line-broker-account
                     (my/beancount--broker-account-alias (match-string 1 trimmed))))
          (setq code (match-string 2 trimmed)
                shares (match-string 3 trimmed)))
         ((string-match "\\`\\([^[:space:],，]+\\)[[:space:],，]+\\(.+\\)\\'" trimmed)
          (when (my/beancount--broker-account-alias (match-string 1 trimmed))
            (user-error "Line %d should be: ACCOUNT CODE SHARES" line-number))
          (setq code (match-string 1 trimmed)
                shares (match-string 2 trimmed)))
         (t
          (user-error "Line %d should be: CODE SHARES or ACCOUNT CODE SHARES" line-number)))
        (let* ((shares (my/beancount--normalize-number shares))
               (instrument (my/beancount--holding-instrument code))
               (is-etf (assoc code my/beancount-funds))
               (account (if is-etf
                            (my/beancount--broker-fund-account
                             instrument (or line-broker-account broker-account))
                          (plist-get instrument :account)))
               (commodity (plist-get instrument :commodity)))
          (when (and line-broker-account (not is-etf))
            (user-error "Line %d has broker account prefix for non-ETF code: %s"
                        line-number code))
          (unless (string-match-p "\\`[-+]?[0-9]+\\(?:\\.[0-9]+\\)?\\'" shares)
            (user-error "Line %d has invalid shares/amount: %s" line-number shares))
          (list :code code :shares shares :account account :commodity commodity))))))

(defun my/beancount--balance-batch-build (date text &optional broker-account)
  "Build a Beancount balance snapshot for DATE from pasted TEXT."
  (let (postings)
    (cl-loop for line in (split-string text "\n")
             for line-number from 1
             for parsed = (my/beancount--parse-balance-batch-line
                           line line-number (or broker-account "普通账户"))
             when parsed
             do (push (format "%s balance %s %s %s"
                              date
                              (plist-get parsed :account)
                              (plist-get parsed :shares)
                              (plist-get parsed :commodity))
                      postings))
    (unless postings
      (user-error "No holdings found. Paste lines like: 513870 1234.56"))
    (list :count (length postings)
          :entry (mapconcat #'identity
                            (cons (format "; ---- Holdings balance snapshot %s ----" date)
                                  (nreverse postings))
                            "\n"))))

(defconst my/beancount-etf-commission-rate 0.00005
  "ETF commission rate for the current broker: 0.5 per 10,000.")

(defconst my/beancount-etf-min-commission 5.0
  "Minimum ETF commission in CNY for the current broker.")

(defun my/beancount--etf-commission (fund trade-value)
  "Return ETF commission for FUND and TRADE-VALUE."
  (if (plist-get fund :fee-free)
      0
    (max my/beancount-etf-min-commission
         (* trade-value my/beancount-etf-commission-rate))))

  "Read a Beancount date string."

(defun my/beancount--safe-string (value)
  "Return VALUE trimmed and safe for a Beancount quoted string."
  (replace-regexp-in-string "\"" "'" (string-trim value)))

(defun my/beancount--insert-before-first-entry (text)
  "Insert TEXT before the first price/transaction/balance entry, else at EOF."
  (goto-char (point-min))
  (if (re-search-forward "^[0-9]\{4\}-[0-9]\{2\}-[0-9]\{2\} \\(price\\|balance\\|[*!]\\)" nil t)
      (progn
        (beginning-of-line)
        (insert text "\n"))
    (goto-char (point-max))
    (unless (bolp) (insert "\n"))
    (insert text "\n")))

(defun my/beancount--ensure-fund-open (date code &optional account)
  "Ensure commodity and account directives for CODE exist, using DATE if inserted.
When ACCOUNT is non-nil, ensure that account instead of the default normal account."
  (let* ((fund (my/beancount--fund code))
         (name (plist-get fund :name))
         (commodity (plist-get fund :commodity))
         (account (or account (plist-get fund :account)))
         (category (plist-get fund :category))
         directives)
    (save-excursion
      (goto-char (point-min))
      (unless (re-search-forward (format "^[0-9-]+ commodity %s\\b" (regexp-quote commodity)) nil t)
        (push (format "%s commodity %s\n  name: \"%s\"\n  code: \"%s\"\n  asset_category: \"%s\""
                      date commodity name code category)
              directives))
      (goto-char (point-min))
      (unless (re-search-forward (format "^[0-9-]+ open %s\\b" (regexp-quote account)) nil t)
        (push (format "%s open %s %s" date account commodity) directives))
      (when directives
        (my/beancount--insert-before-first-entry
         (mapconcat #'identity (nreverse directives) "\n\n"))))))

(defun my/beancount--append-entry (entry &optional date code account)
  "Append ENTRY to main ledger, optionally ensuring CODE is open at DATE."
  (find-file (my/beancount-main-file))
  (when (and date code)
    (my/beancount--ensure-fund-open date code account))
  (goto-char (point-max))
  (unless (bolp) (insert "\n"))
  (insert "\n" entry)
  (save-buffer)
  (my/beancount-check-main))

(defun my/beancount-python-executable ()
  "Return a Python executable with the investment script dependencies.
Linux 优先用 ~/.venvs/beancount（uv 建的 venv，装了 beancount/fava 全套）；
Windows 用 scoop python311；都找不到才退回 PATH 上的 python3/python。"

                          (expand-file-name "scoop/apps/python311/current/python.exe" "~")
                          (executable-find "python3")
                          (executable-find "python"))))
    (catch 'found
      (dolist (candidate candidates)
        (when (and candidate (file-executable-p candidate))
          (throw 'found candidate)))
      "python")))

(defun my/beancount-script (name)
  "Resolve the absolute path of a Beancount console script NAME (e.g. \"bean-check\", \"fava\").
The scoop Python that installs beancount/fava puts its Scripts directory
(scoop/persist/python311/Scripts) outside PATH, so `executable-find' alone fails.
Fall back to the script sitting next to `my/beancount-python-executable'."
  (or (executable-find name)
      (let* ((py (my/beancount-python-executable))
             (py-dir (and (file-name-absolute-p py)
                          (file-name-directory (expand-file-name py))))
             (cands (when py-dir
                      (list (expand-file-name (concat "Scripts/" name ".exe") py-dir)
                            (expand-file-name (concat "Scripts/" name) py-dir)
                            (expand-file-name name py-dir)))))
        (catch 'found
          (dolist (cand cands)
            (when (file-exists-p cand) (throw 'found cand)))
          name))))
